const int PWM_CHANNEL = 0;
const int PWM_FREQ    = 5000;
const int PWM_BITS    = 8;

WebServer server(80);
int brightness = 50;

void applyBrightness(int pct) {
  pct = constrain(pct, 0, 100);
  ledcWrite(PWM_CHANNEL, map(pct, 0, 100, 0, 255));
  Serial.printf("[LED] %d%%\n", pct);
}

void handleRoot() {
  String page = String(HTML);
  page.replace("__BR__", String(brightness));
  server.send(200, "text/html", page);
}

void handleSet() {
  if (server.hasArg("brightness")) {
    brightness = server.arg("brightness").toInt();
    applyBrightness(brightness);
    server.send(200, "text/plain", "OK");
  } else {
    server.send(400, "text/plain", "Missing param");
  }
}

void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println("\n=== ESP32 LED Controller ===");
  ledcSetup(PWM_CHANNEL, PWM_FREQ, PWM_BITS);
  ledcAttachPin(LED_PIN, PWM_CHANNEL);
  applyBrightness(brightness);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.printf("Connecting to %s", WIFI_SSID);
  int tries = 0;
  while (WiFi.status() != WL_CONNECTED) {
    delay(500); Serial.print(".");
    if (++tries > 40) {
      Serial.println("\nFailed! Restarting...");
      delay(3000); ESP.restart();
    }
  }
  Serial.println("\nConnected!");
  Serial.printf("Open browser: http://%s\n",
                WiFi.localIP().toString().c_str());
  server.on("/",    HTTP_GET, handleRoot);
  server.on("/set", HTTP_GET, handleSet);
  server.onNotFound([](){
    server.send(404, "text/plain", "Not found");
  });
  server.begin();
  Serial.println("Server started!");
}

void loop() {
  server.handleClient();
  delay(2);
}

PHASE 04

Wire the Hardware

HARDWARE
08
Connect LED to ESP32
3 connections total — resistor protects the LED
FromViaToNote
GPIO 2 → 220Ω resistor → LED long leg (+) Always use resistor
LED short leg (−) → wire → GND Any GND pin
💡No breadboard? GPIO2 is already connected to the built-in blue LED on most ESP32 boards. Just plug in USB and test — no wiring needed.

IoT-based LED brightness control system using ESP32-WROOM.

- ESP32-WROOM board
- 1x LED
- 1x 220Ω resistor

GPIO2 → 220Ω resistor → LED(+) → LED(-) → GND

- Web-based slider UI
